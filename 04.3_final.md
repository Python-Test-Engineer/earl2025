import pandas as pd
import numpy as np

def impute_missing_values_with_mean(df: pd.DataFrame) -> pd.DataFrame:
    """
    Impute missing values in a Pandas DataFrame with the mean of the column.

    Args:
    - df (pd.DataFrame): The input DataFrame.

    Returns:
    - df_imputed (pd.DataFrame): The DataFrame with missing values imputed.
    """
    # Check if input is a Pandas DataFrame
    if not isinstance(df, pd.DataFrame):
        raise ValueError("Input must be a Pandas DataFrame")

    # Create a copy of the original DataFrame to avoid modifying it
    df_imputed = df.copy()

    # Check for missing values
    if not df_imputed.isnull().values.any():
        return df_imputed  # Return original DataFrame if no missing values

    # Impute missing values using apply method
    numerical_cols = df_imputed.select_dtypes(include=[np.number]).columns
    df_imputed[numerical_cols] = df_imputed[numerical_cols].apply(lambda x: x.fillna(x.mean(skipna=True)))

    return df_imputed

# Example usage
if __name__ == "__main__":
    try:

        }
        df = pd.DataFrame(data)
