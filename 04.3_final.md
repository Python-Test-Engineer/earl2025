import pandas as pd
import numpy as np

def impute_missing_values_with_mean(df):
    """
    Imputes missing values in a DataFrame with the mean of each numeric column.

    Parameters:
    df (pd.DataFrame): The input DataFrame with possible missing values.

    Returns:
    pd.DataFrame: A new DataFrame where missing values in numeric columns are replaced with their respective column means.

    Raises:
    ValueError: If the input is not a DataFrame, if the DataFrame is empty, or if no numeric columns are present.
    """
    if not isinstance(df, pd.DataFrame):
        raise ValueError("Input must be a pandas DataFrame.")
    if df.empty:
        raise ValueError("Input DataFrame is empty.")

    # Create a copy of the DataFrame to avoid modifying the original
    df_imputed = df.copy()

    # Select only numeric columns for imputation
    numeric_cols = df_imputed.select_dtypes(include=[np.number]).columns
    
    if numeric_cols.empty:
        raise ValueError("No numeric columns available for imputation.")

    # Impute missing values with the mean for each numeric column
    df_imputed[numeric_cols] = df_imputed[numeric_cols].fillna(df_imputed[numeric_cols].mean())

    return df_imputed

# Example usage
if __name__ == "__main__":
    data = {
        'A': [1, 2, np.nan, 4],
        'B': [np.nan, 2, 3, 4],
        'C': [1, 2, 3, 'missing']  # Non-numeric column example
    }
    df = pd.DataFrame(data)

    print("Original DataFrame:")
    print(df)

    df_imputed = impute_missing_values_with_mean(df)

    print("\nDataFrame after imputing missing values with mean:")
    print(df_imputed)
