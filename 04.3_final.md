import pandas as pd
import numpy as np

def impute_missing_values(df: pd.DataFrame) -> pd.DataFrame:
    """
    Imputes missing values in a Pandas DataFrame with the mean of the column.

    Args:
        df (pd.DataFrame): Input DataFrame.

    Returns:
        pd.DataFrame: DataFrame with missing values imputed.

    Raises:
        ValueError: If the input is not a Pandas DataFrame or if a column contains non-numeric values.
    """
    try:
        # Check if the input is a Pandas DataFrame
        if not isinstance(df, pd.DataFrame):
            raise ValueError("Input must be a Pandas DataFrame")

        # Check if the DataFrame contains at least one column
        if len(df.columns) == 0:
            raise ValueError("DataFrame must contain at least one column")

        # Create a copy of the input DataFrame to avoid modifying it in-place
        imputed_df = df.copy()

        # Iterate over each column in the DataFrame
        for column in imputed_df.columns:
            # Check if the column contains non-numeric values
            if not pd.api.types.is_numeric_dtype(imputed_df[column]):
                print(f"Skipping non-numeric column: {column}")
                continue

            # Check if the column contains any missing values
            if imputed_df[column].isnull().any():
                # Calculate the mean of the column, excluding missing values
                column_mean = imputed_df[column].mean()

                # Replace missing values in the column with the calculated mean
                imputed_df[column] = imputed_df[column].fillna(column_mean)

        return imputed_df

    except ValueError as e:
        print(f"Error: {e}")
        return None

    except Exception as e:
        print(f"An unexpected error occurred: {e}")
        return None
