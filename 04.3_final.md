import pandas as pd
import numpy as np

def impute_missing_values(df: pd.DataFrame, method: str = 'mean') -> pd.DataFrame:
    """
    Impute missing values in a DataFrame using the specified method.

    Parameters:
    df (pd.DataFrame): A DataFrame with potential missing values.
    method (str): The method of imputation. Options are 'mean', 'median', or 'mode'.

    Returns:
    pd.DataFrame: A DataFrame with missing values imputed based on the specified method.
    """
    
    df_imputed = df.copy()  # Create a copy of the DataFrame to avoid in-place modification
    numeric_columns = df_imputed.select_dtypes(include=[np.number]).columns.tolist()

    if len(numeric_columns) == 0:
        raise ValueError("No numeric columns available for imputation.")

    # Impute missing values based on the specified method
    if method == 'mean':
        means = df_imputed[numeric_columns].mean(numeric_only=True)
        df_imputed[numeric_columns] = df_imputed[numeric_columns].fillna(means)
    elif method == 'median':
        medians = df_imputed[numeric_columns].median(numeric_only=True)
        df_imputed[numeric_columns] = df_imputed[numeric_columns].fillna(medians)
    elif method == 'mode':
        modes = df_imputed[numeric_columns].mode().iloc[0]  # Get the first mode if there are multiple
        df_imputed[numeric_columns] = df_imputed[numeric_columns].fillna(modes)
    else:
        raise ValueError("Invalid method specified. Use 'mean', 'median', or 'mode'.")

    return df_imputed

# Example usage
if __name__ == "__main__":
    # Creating a sample DataFrame with missing values
    data = {
        'A': [1, 2, np.nan, 4],
        'B': [np.nan, 5, 6, 7],
        'C': ['a', 'b', 'c', 'd']  # C is a non-numeric column
    }
    df = pd.DataFrame(data)

    print("Original DataFrame:")
    print(df)

    imputed_df = impute_missing_values(df, method='mean')
    
    print("\nDataFrame after mean imputation:")
    print(imputed_df)
