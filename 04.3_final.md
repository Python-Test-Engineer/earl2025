import pandas as pd
import numpy as np

def impute_missing_values(df, inplace=False):
    """
    Imputes missing values in the DataFrame with the mean of each numeric column.
    
    Parameters:
    df (pd.DataFrame): The DataFrame with potential missing values. Non-numeric columns will be skipped.
    inplace (bool): If True, modifies the DataFrame in place; otherwise, returns a new DataFrame.
    
    Returns:
    pd.DataFrame or None: A new DataFrame with missing values imputed if inplace is False,
                          otherwise returns None.
    """
    # Ensure to work with a copy to avoid modifying the input directly (if not inplace)
    if not inplace:
        df = df.copy()

    # Check for numeric columns and calculate the mean
    numeric_cols = df.select_dtypes(include=[np.number])
    if numeric_cols.empty:
        print("Warning: No numeric columns found to impute.")
        return df if not inplace else None

    means = numeric_cols.mean()

    # Impute missing values with the mean only for numeric columns
    df.fillna(means, inplace=True)

    # Return None if in-place modification is intended
    if inplace:
        return None
    return df

# Example usage:
if __name__ == "__main__":
    # Creating a sample DataFrame with missing values and a non-numeric column
    data = {

        'C': ['string', 'test', np.nan, 'example']  # Non-numeric
    }
    
    df = pd.DataFrame(data)
    print("Original DataFrame:")
    print(df)
    
    # Impute missing values
    imputed_df = impute_missing_values(df)
    print("\nDataFrame after imputing missing values:")
    print(imputed_df)

    # Demonstrating in-place modification
