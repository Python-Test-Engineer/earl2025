def impute_missing_values_vectorized(df: pd.DataFrame) -> pd.DataFrame:
    """
    Imputes missing values in a Pandas DataFrame with the mean of the column using vectorized operations.

    Args:
        df (pd.DataFrame): Input DataFrame.

    Returns:
        pd.DataFrame: DataFrame with missing values imputed.
    """
    try:
        # Check if the input is a Pandas DataFrame
        if not isinstance(df, pd.DataFrame):
            raise ValueError("Input must be a Pandas DataFrame")

        # Check if the DataFrame contains at least one column
        if len(df.columns) == 0:
            raise ValueError("DataFrame must contain at least one column")

        # Create a copy of the input DataFrame to avoid modifying it in-place
        imputed_df = df.copy()

        # Separate numeric columns
        numeric_df = imputed_df.select_dtypes(include=[np.number])

        # Calculate the mean of each numeric column, excluding missing values
        means = numeric_df.mean()

        # Replace missing values in the numeric columns with the calculated means
        imputed_numeric_df = numeric_df.fillna(means)

        # Combine the imputed numeric columns with the non-numeric columns
        imputed_df[list(numeric_df.columns)] = imputed_numeric_df

        return imputed_df
