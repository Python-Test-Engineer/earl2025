import pandas as pd
import numpy as np
from typing import Optional

def impute_missing_values(df: pd.DataFrame, imputation_method: str = 'mean', ignore_non_numeric: bool = True, custom_imputation_value: Optional[float] = None, custom_non_numeric_imputation_value: Optional[str] = None) -> pd.DataFrame:
    """
    Impute missing values in a Pandas DataFrame.

    Parameters:
    df (pd.DataFrame): DataFrame with missing values.
    imputation_method (str): Method to use for imputation. Can be 'mean', 'median', 'mode', or 'custom'. Default is 'mean'.
    ignore_non_numeric (bool): Whether to ignore non-numeric columns. Default is True.
    custom_imputation_value (float): Custom value to use for imputation if imputation_method is 'custom'. Default is None.
    custom_non_numeric_imputation_value (str): Custom value to use for imputation of non-numeric columns if ignore_non_numeric is False. Default is None.

    Returns:
    pd.DataFrame: DataFrame with missing values imputed.

    Raises:
    ValueError: If input is not a Pandas DataFrame, if input DataFrame is empty, if custom imputation value is not provided for 'custom' imputation method, or if custom non-numeric imputation value is not provided for non-numeric columns.
    """

    try:
        # Check if input is a Pandas DataFrame
        if not isinstance(df, pd.DataFrame):
            raise ValueError("Input must be a Pandas DataFrame")

        # Check if input DataFrame is empty
        if df.empty:
            raise ValueError("Input DataFrame is empty")

        # Iterate over each column in the DataFrame
        for col in df.columns:
            # Check if column contains numeric values
            if pd.api.types.is_numeric_dtype(df[col]):
                # Impute missing values based on the specified imputation method
                if imputation_method == 'mean':
                    imputation_value = df[col].mean()
                elif imputation_method == 'median':
                    imputation_value = df[col].median()
                elif imputation_method == 'mode':
                    imputation_value = df[col].mode().iloc[0]
                elif imputation_method == 'custom':
                    if custom_imputation_value is None:
                        raise ValueError("Custom imputation value must be provided for 'custom' imputation method")
                    imputation_value = custom_imputation_value
                else:
                    raise ValueError("Invalid imputation method")

                # Impute missing values with the calculated imputation value
                df[col] = df[col].fillna(imputation_value)
            else:
                # Handle non-numeric columns based on the ignore_non_numeric parameter
                if ignore_non_numeric:
                    print(f"Column '{col}' is not numeric and will be ignored.")
                else:
                    if custom_non_numeric_imputation_value is None:
                        raise ValueError(f"Custom non-numeric imputation value must be provided for column '{col}'")
                    df[col] = df[col].fillna(custom_non_numeric_imputation_value)

        # Return the DataFrame with missing values imputed
        return df

    except ValueError as e:
        print(f"Error: {e}")
        return None
    except Exception as e:
        print(f"An unexpected error occurred: {e}")
        return None
