import pandas as pd
import numpy as np

def impute_missing_values_with_mean(df: pd.DataFrame) -> pd.DataFrame:
    """
    Impute missing values in a DataFrame with the mean of each column.

    Parameters:
    df (pd.DataFrame): The DataFrame with potentially missing values.

    Returns:
    pd.DataFrame: A DataFrame with missing values imputed using column means.
    """
    
    df_imputed = df.copy()  # Avoid modifying in place
    numeric_columns = df_imputed.select_dtypes(include=[np.number]).columns

    # Fill missing values with the mean for each numeric column
    df_imputed[numeric_columns] = df_imputed[numeric_columns].fillna(df_imputed.mean(numeric_only=True))
    
    return df_imputed
