import pandas as pd
import numpy as np
from typing import Optional

def impute_missing_values(df: pd.DataFrame, imputation_method: str = 'mean', ignore_non_numeric: bool = True, custom_imputation_value: Optional[float] = None) -> pd.DataFrame:
    """
    Impute missing values in a Pandas DataFrame.

    Parameters:
    df (pd.DataFrame): DataFrame with missing values.
    imputation_method (str): Method to use for imputation. Can be 'mean', 'median', 'mode', or 'custom'. Default is 'mean'.
    ignore_non_numeric (bool): Whether to ignore non-numeric columns. Default is True.
    custom_imputation_value (float): Custom value to use for imputation if imputation_method is 'custom'. Default is None.

    Returns:
    pd.DataFrame: DataFrame with missing values imputed.
    """
    if not isinstance(df, pd.DataFrame):
        raise ValueError("Input must be a Pandas DataFrame")

    if df.empty:
        raise ValueError("Input DataFrame is empty")

    for col in df.columns:
        if pd.api.types.is_numeric_dtype(df[col]):
            if imputation_method == 'mean':
                imputation_value = df[col].mean()
            elif imputation_method == 'median':
                imputation_value = df[col].median()
            elif imputation_method == 'mode':
                imputation_value = df[col].mode().iloc[0]
            elif imputation_method == 'custom':
                if custom_imputation_value is None:
                    raise ValueError("Custom imputation value must be provided for 'custom' imputation method")
                imputation_value = custom_imputation_value
            else:
                raise ValueError("Invalid imputation method")

            df[col] = df[col].fillna(imputation_value)
        else:
            if ignore_non_numeric:
                print(f"Column '{col}' is not numeric and will be ignored.")
            else:
                if custom_imputation_value is not None:
                    df[col] = df[col].fillna(custom_imputation_value)
                else:
                    print(f"Column '{col}' is not numeric and cannot be imputed. Please provide a custom imputation value.")

    return df

# Example usage
if __name__ == "__main__":
    data = {
        "A": [1, 2, np.nan, 4, 5],
        "B": [np.nan, 2, 3, 4, 5],
        "C": ["a", "b", np.nan, "d", "e"]  # Non-numeric column
    }
    df = pd.DataFrame(data)

    print("Original DataFrame:")
    print(df)

    df_imputed = impute_missing_values(df, imputation_method='mean')

    print("\nDataFrame with missing values imputed:")
    print(df_imputed)
