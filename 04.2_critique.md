```python
import pandas as pd
import numpy as np

def impute_missing_values(df, inplace=False):
    """
    Imputes missing values in the DataFrame with the mean of each numeric column.
    
    Parameters:
    df (pd.DataFrame): The DataFrame with potential missing values.
    inplace (bool): If True, modifies the DataFrame in place; otherwise, returns a new DataFrame.
    
    Returns:
    pd.DataFrame: A new DataFrame with missing values imputed if inplace is False.
    """
    # Ensure we work with a copy to avoid modifying the input directly (if not inplace)
    if not inplace:
        df = df.copy()
    
    # Calculate the mean for each numeric column
    means = df.select_dtypes(include=[np.number]).mean()
    
    # Impute missing values with the mean only for numeric columns
    imputed_df = df.fillna(means)
    
    if not inplace:
        return imputed_df

# Example usage:
if __name__ == "__main__":
    # Creating a sample DataFrame with missing values and a non-numeric column
    data = {
        'A': [1, 2, np.nan, 4],
        'B': [np.nan, 5, 6, 7],
        'C': ['string', 'test', np.nan, 'example']  # Non-numeric
    }
    
    df = pd.DataFrame(data)
    print("Original DataFrame:")
    print(df)
    
    # Impute missing values
    imputed_df = impute_missing_values(df)
    print("\nDataFrame after imputing missing values:")
    print(imputed_df)
```
