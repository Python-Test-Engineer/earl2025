import pandas as pd
import numpy as np

# Creating a sample DataFrame for demonstration
data = {
    'A': [1, 2, np.nan, 4],
    'B': [np.nan, 2, 3, 4],
    'C': [1, np.nan, np.nan, 4],
    'D': ['a', 'b', 'c', 'd']  # Non-numeric column for testing
}

df = pd.DataFrame(data)

# Display the original DataFrame
print("Original DataFrame:")
print(df)

def impute_missing_with_mean(dataframe, inplace=False):
    """
    Impute missing values in the DataFrame with the mean of each column.
    
    Parameters:
        dataframe (pd.DataFrame): The DataFrame to process.
        inplace (bool): If True, perform in-place modification, else return a new DataFrame.

    Returns:
        pd.DataFrame or None: A new DataFrame with missing values filled by column means if inplace=False. None if inplace=True.
    """
    numeric_columns = dataframe.select_dtypes(include=[np.number]).columns
    
    if inplace:
        dataframe[numeric_columns] = dataframe[numeric_columns].fillna(dataframe[numeric_columns].mean())
        # Returning None when doing in-place modification
        return None  
    else:
        new_df = dataframe.copy()  # Ensure we do not modify the original DataFrame
        new_df[numeric_columns] = new_df[numeric_columns].fillna(new_df[numeric_columns].mean())
        return new_df

# Example: In-place imputation
impute_missing_with_mean(df, inplace=True)
print("\nDataFrame After In-Place Imputation:")
print(df)

# Example: Returning a new DataFrame
df_new_imputed = impute_missing_with_mean(df)
print("\nNew DataFrame After Imputation (Returned):")
print(df_new_imputed)
